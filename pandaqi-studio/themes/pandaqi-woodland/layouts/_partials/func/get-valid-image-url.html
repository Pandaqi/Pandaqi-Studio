{{- $src := .src -}} {{- range slice ".webp" ".jpg" ".jpeg" ".png" -}} {{- $src = strings.TrimSuffix . $src -}} {{- end -}} {{- $page_url := .context.Page.RelPermalink -}} {{- $final_url := "" -}} {{- range slice "webp" "jpg" "jpeg" "png" -}} {{- $candidate := printf "content%s%s.%s" $page_url $src . -}} {{- if and (not $final_url) (fileExists $candidate) -}} {{- $final_url = strings.TrimPrefix "content" $candidate -}} {{- end -}} {{- end -}} {{- return $final_url -}}
