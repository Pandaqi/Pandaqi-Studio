{{- $subdir := .collection -}} {{- $final_pages := slice -}} {{- if $subdir -}} {{- $url := trim .context.RelPermalink "/" -}} {{- $final_dir := printf "content/%s/%s/" $url $subdir -}} {{- $pages := union (where .context.Pages "Type" "project") (where .context.Pages "Type" "tools") -}} {{- if (fileExists $final_dir) -}} {{- range (readDir $final_dir) -}} {{- if .IsDir -}} {{- $path1 := printf "%s/%s/%s/" $url $subdir .Name -}} {{- $path2 := replace $path1 "/" "\\" -}} {{- $selected_pages := union (where $pages ".File.Dir" $path1) (where $pages ".File.Dir" $path2) -}} {{- $final_pages = $final_pages | union $selected_pages -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- return $final_pages -}}
